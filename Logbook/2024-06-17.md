```ts
// eslint-disable-next-line class-methods-use-this
mdEventListen: Function = () => {};
  
closeColumnChooser(e: any) {
	const button = e.target.closest('.dx-datagrid-column-chooser-button');
	const columnChooser = e.target.closest('.dx-datagrid-column-chooser');
	if (!button && !columnChooser)
		this.dataGrid?.instance.hideColumnChooser();
}

gridInit(event: any) {
	this.mdEventListen = this.renderer.listen(
		'document',
		'mousedown',
		this.closeColumnChooser.bind(this)
	);
}

ngOnDestroy() {
	this.mdEventListen!();
}
```
